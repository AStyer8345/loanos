import type { LoanPurpose, LoanType, PropertyType } from './types'

const LOAN_PURPOSE_MAP: Record<string, LoanPurpose> = {
  Purchase: 'purchase',
  Refinance: 'refinance',
  CashOutRefinance: 'cash_out_refi',
}

const LOAN_TYPE_MAP: Record<string, LoanType> = {
  Conventional: 'conventional',
  FHA: 'fha',
  VA: 'va',
  USDARuralDevelopment: 'usda',
}

const PROPERTY_TYPE_MAP: Record<string, PropertyType> = {
  PrimaryResidence: 'primary_residence',
  SecondHome: 'second_home',
  Investment: 'investment',
}

export function normalizeLoanPurpose(v: string): LoanPurpose | null {
  return LOAN_PURPOSE_MAP[v] ?? null
}

export function normalizeLoanType(v: string): LoanType | null {
  return LOAN_TYPE_MAP[v] ?? null
}

export function normalizePropertyType(v: string): PropertyType | null {
  return PROPERTY_TYPE_MAP[v] ?? null
}

export function normalizePhone(raw: string): string | null {
  const digits = raw.replace(/\D/g, '')
  // Accept 10 digits, or 11 digits starting with 1 (US country code)
  if (digits.length === 10) return digits
  if (digits.length === 11 && digits.startsWith('1')) return digits.slice(1)
  return null
}

export function parseCurrency(raw: string): number | null {
  if (!raw) return null
  const n = Number(raw)
  return Number.isFinite(n) ? n : null
}

export function parseIsoDate(raw: string): string | null {
  if (!raw) return null
  const d = new Date(raw)
  return Number.isFinite(d.getTime()) ? raw : null
}

export function normalizeAddressForMatch(raw: string): string {
  return raw
    .toLowerCase()
    .replace(/[^\w\s]/g, ' ')
    .replace(/\s+/g, ' ')
    .trim()
}

export function normalizeEmailForMatch(raw: string): string {
  return raw.toLowerCase().trim()
}

export function normalizeLastNameForMatch(raw: string): string {
  return raw.normalize('NFC').toLowerCase().trim()
}
```

import { describe, it, expect } from 'vitest'
import { readFileSync, writeFileSync, existsSync } from 'node:fs'
import { join } from 'node:path'
import { parseMismo } from '@/lib/mismo/parser'
import type { ParsedMismo } from '@/lib/mismo/types'

const FIXTURES = join(__dirname, 'fixtures')
const UPDATE = process.env.UPDATE_SNAPSHOTS === '1'

function loadFixture(name: string): string {
  return readFileSync(join(FIXTURES, name), 'utf-8')
}

function snapshotPath(name: string): string {
  return join(FIXTURES, `${name.replace(/\.xml$/, '')}.expected.json`)
}

function checkOrUpdateSnapshot(fixtureName: string, actual: ParsedMismo) {
  const path = snapshotPath(fixtureName)
  if (UPDATE || !existsSync(path)) {
    writeFileSync(path, JSON.stringify(actual, null, 2) + '\n', 'utf-8')
    return
  }
  const expected = JSON.parse(readFileSync(path, 'utf-8'))
  expect(actual).toEqual(expected)
}

describe('parseMismo — happy paths', () => {
  it('parses a purchase fixture', () => {
    const result = parseMismo(loadFixture('sample-purchase.xml'))
    expect(result.ok).toBe(true)
    if (!result.ok) throw new Error('expected ok result')
    expect(result.data.loan.loan_purpose).toBe('purchase')
    expect(result.data.borrower.last_name).toBe('Doe')
    expect(result.data.property.city).toBe('Austin')
    checkOrUpdateSnapshot('sample-purchase.xml', result.data)
  })

  it('parses a refi fixture', () => {
    const result = parseMismo(loadFixture('sample-refi.xml'))
    expect(result.ok).toBe(true)

    checkOrUpdateSnapshot('sample-refi.xml', result.data)

    if (!result.ok) throw new Error('expected ok result')

import { XMLParser } from 'fast-xml-parser'
import type {
  ParsedMismo,
  ParsedBorrower,
  ParsedProperty,
  ParsedLoanTerms,
  ParseResult,
  ParseError,
} from './types'
import {
  normalizeLoanPurpose,
  normalizeLoanType,
  normalizePropertyType,
  normalizePhone,
  parseCurrency,
  parseIsoDate,
} from './normalize'

const xmlParser = new XMLParser({
  ignoreAttributes: false,
  attributeNamePrefix: '@_',
  allowBooleanAttributes: true,
  parseTagValue: false,
  parseAttributeValue: false,
  trimValues: true,
})

export function parseMismo(xml: string): ParseResult {
  if (!xml.trim().startsWith('<')) {
    return err({ code: 'not_xml', message: 'Input does not appear to be XML' })
  }

  let doc: Record<string, unknown>
  try {
    doc = xmlParser.parse(xml) as Record<string, unknown>
  } catch (e) {
    return err({ code: 'malformed_xml', message: (e as Error).message })
  }

  const message = doc['MESSAGE'] as Record<string, unknown> | undefined
  if (!message || typeof message !== 'object') {
    return err({ code: 'not_mismo_3_4', message: 'Root MESSAGE element not found' })
  }

  const version = getMismoVersion(message)
  if (version !== '3.4') {
    return err({
      code: 'not_mismo_3_4',
      message: `Expected MISMO 3.4, found ${version ?? 'unknown'}`,
      detectedVersion: version ?? undefined,
    })
  }

  const deal = extractDeal(message)
  if (!deal) return err({ code: 'not_mismo_3_4', message: 'DEAL element not found' })

  const parties = getArray(deal, 'PARTIES.PARTY') as Array<Record<string, unknown>>
  const borrowers = parties.filter(isBorrowerParty)
  if (borrowers.length === 0) return err({ code: 'missing_required', message: 'No borrower party found', missingFields: ['borrower'] })

  const borrower = extractBorrower(borrowers[0])
  const coBorrower = borrowers[1] ? extractBorrower(borrowers[1]) : null

  const property = extractProperty(deal)
  const loan = extractLoan(deal)

  const missing = collectMissingRequired({ borrower, property, loan })
  if (missing.length > 0) {
    return err({ code: 'missing_required', message: `Missing required fields: ${missing.join(', ')}`, missingFields: missing })
  }

  return {
    ok: true,
    data: {
      borrower,
      co_borrower: coBorrower,
      property: property as ParsedProperty, // validated in collectMissingRequired
      loan: loan as ParsedLoanTerms,
    },
  }
}

function err(error: ParseError): ParseResult {
  return { ok: false, error }
}

function getMismoVersion(message: Record<string, unknown>): string | null {
  const version = (message['@_MISMOReferenceModelIdentifier'] as string | undefined) ?? null
  if (version?.startsWith('3.4')) return '3.4'
  return version
}

function extractDeal(message: Record<string, unknown>): Record<string, unknown> | null {
  const dealSets = getPath(message, 'DEAL_SETS.DEAL_SET') as Record<string, unknown> | undefined
  if (!dealSets) return null
  const deals = getPath(dealSets, 'DEALS.DEAL') as Record<string, unknown> | undefined
  return deals ?? null
}

function isBorrowerParty(party: Record<string, unknown>): boolean {
  const roleType = getPath(party, 'ROLES.ROLE.ROLE_DETAIL.PartyRoleType') as string | undefined
  return roleType === 'Borrower'
}

function extractBorrower(party: Record<string, unknown>): ParsedBorrower {
  const individual = getPath(party, 'INDIVIDUAL') as Record<string, unknown> | undefined
  const firstName = getPath(individual ?? {}, 'NAME.FirstName') as string | undefined
  const lastName = (getPath(individual ?? {}, 'NAME.LastName') as string | undefined) ?? ''
  const dob = getPath(individual ?? {}, 'INDIVIDUAL_DETAIL.BirthDate') as string | undefined

  const contactPoints = getArray(individual ?? {}, 'CONTACT_POINTS.CONTACT_POINT') as Array<Record<string, unknown>>
  const email = findContactValue(contactPoints, 'Email') ?? null
  const phone = findContactValue(contactPoints, 'Phone') ?? null

  const addresses = getArray(party, 'ADDRESSES.ADDRESS') as Array<Record<string, unknown>>
  const current = addresses.find(a => (a['AddressType'] as string) === 'Current') ?? addresses[0]
  const mailingAddress = current ? formatAddress(current) : null

  return {
    first_name: firstName ?? null,
    last_name: lastName,
    email,
    phone: phone ? normalizePhone(phone) : null,
    dob: dob ? parseIsoDate(dob) : null,
    mailing_address: mailingAddress,
  }
}

function findContactValue(points: Array<Record<string, unknown>>, type: 'Email' | 'Phone'): string | null {
  for (const p of points) {
    const detail = p['CONTACT_POINT_DETAIL'] as Record<string, unknown> | undefined
    const roleType = detail?.['ContactPointRoleType'] as string | undefined
    if (type === 'Email') {
      const emailObj = p['CONTACT_POINT_EMAIL'] as Record<string, unknown> | undefined
      const value = emailObj?.['ContactPointEmailValue'] as string | undefined
      if (value) return value
    } else {
      const phoneObj = p['CONTACT_POINT_TELEPHONE'] as Record<string, unknown> | undefined
      const value = phoneObj?.['ContactPointTelephoneValue'] as string | undefined
      if (value && (roleType === 'Mobile' || roleType === 'Home' || !roleType)) return value
    }
  }
  return null
}

function formatAddress(a: Record<string, unknown>): string {
  const line = (a['AddressLineText'] as string | undefined) ?? ''
  const city = (a['CityName'] as string | undefined) ?? ''
  const state = (a['StateCode'] as string | undefined) ?? ''
  const zip = (a['PostalCode'] as string | undefined) ?? ''
  return [line, city, state, zip].filter(Boolean).join(', ')
}

function extractProperty(deal: Record<string, unknown>): Partial<ParsedProperty> {
  const collateral = getPath(deal, 'COLLATERALS.COLLATERAL') as Record<string, unknown> | undefined
  if (!collateral) return {}
  const subject = getPath(collateral, 'SUBJECT_PROPERTY') as Record<string, unknown> | undefined
  if (!subject) return {}

  const addr = getPath(subject, 'ADDRESS') as Record<string, unknown> | undefined
  const detail = getPath(subject, 'PROPERTY_DETAIL') as Record<string, unknown> | undefined
  const propertyUsage = detail?.['PropertyUsageType'] as string | undefined

  const salesContract = getPath(subject, 'SALES_CONTRACTS.SALES_CONTRACT.SALES_CONTRACT_DETAIL.SalesContractAmount') as string | undefined
  const valuation = getPath(subject, 'PROPERTY_VALUATIONS.PROPERTY_VALUATION.PROPERTY_VALUATION_DETAIL.PropertyValueEstimateAmount') as string | undefined

  const propertyValue = salesContract ? parseCurrency(salesContract) : (valuation ? parseCurrency(valuation) : null)

  return {
    address: (addr?.['AddressLineText'] as string | undefined) ?? undefined,
    city: (addr?.['CityName'] as string | undefined) ?? undefined,
    state: (addr?.['StateCode'] as string | undefined) ?? undefined,
    zip: (addr?.['PostalCode'] as string | undefined) ?? undefined,
    property_type: propertyUsage ? normalizePropertyType(propertyUsage) : null,
    property_value: propertyValue,
  }
}

function extractLoan(deal: Record<string, unknown>): Partial<ParsedLoanTerms> {
  const loans = getPath(deal, 'LOANS.LOAN') as Record<string, unknown> | undefined
  if (!loans) return {}

  const detail = getPath(loans, 'LOAN_DETAIL') as Record<string, unknown> | undefined
  const terms = getPath(loans, 'TERMS_OF_LOAN') as Record<string, unknown> | undefined
  const maturity = getPath(loans, 'MATURITY') as Record<string, unknown> | undefined
  const closing = getPath(loans, 'CLOSING_INFORMATION.CLOSING_INFORMATION_DETAIL.ClosingDate') as string | undefined

  const identifiers = getArray(loans, 'LOAN_IDENTIFIERS.LOAN_IDENTIFIER') as Array<Record<string, unknown>>
  const lenderLoanId = identifiers.find(i => i['LoanIdentifierType'] === 'LenderLoan')
  const loanNumber = (lenderLoanId?.['LoanIdentifier'] as string | undefined) ?? null

  const loanAmount = terms?.['BaseLoanAmount'] as string | undefined
  const loanPurpose = detail?.['LoanPurposeType'] as string | undefined
  const mortgageType = detail?.['MortgageType'] as string | undefined
  const termMonths = maturity?.['LoanAmortizationPeriodCount'] as string | undefined

  return {
    loan_number: loanNumber,
    loan_amount: loanAmount ? parseCurrency(loanAmount) ?? undefined : undefined,
    loan_purpose: loanPurpose ? normalizeLoanPurpose(loanPurpose) ?? undefined : undefined,
    loan_type: mortgageType ? normalizeLoanType(mortgageType) : null,
    term_months: termMonths ? Number(termMonths) : null,
    estimated_close_date: closing ? parseIsoDate(closing) : null,
  }
}

function collectMissingRequired(p: {
  borrower: ParsedBorrower
  property: Partial<ParsedProperty>
  loan: Partial<ParsedLoanTerms>
}): string[] {
  const missing: string[] = []
  if (!p.borrower.last_name) missing.push('borrower.last_name')
  if (!p.borrower.email && !p.borrower.phone) missing.push('borrower.email_or_phone')
  if (!p.property.address) missing.push('property.address')
  if (!p.property.city) missing.push('property.city')
  if (!p.property.state) missing.push('property.state')
  if (!p.property.zip) missing.push('property.zip')
  if (p.loan.loan_amount == null) missing.push('loan.loan_amount')
  if (!p.loan.loan_purpose) missing.push('loan.loan_purpose')
  return missing
}

function getPath(obj: Record<string, unknown>, path: string): unknown {
  return path.split('.').reduce<unknown>((acc, key) => {
    if (acc == null || typeof acc !== 'object') return undefined
    return (acc as Record<string, unknown>)[key]
  }, obj)
}

function getArray(obj: Record<string, unknown>, path: string): unknown[] {
  const v = getPath(obj, path)
  if (v == null) return []
  return Array.isArray(v) ? v : [v]
}

import type { SupabaseClient } from '@supabase/supabase-js'
import type { ParsedMismo, ContactMatch, LoanMatch } from './types'
import {
  normalizeAddressForMatch,
  normalizeEmailForMatch,
  normalizeLastNameForMatch,
} from './normalize'

// NOTE: the spec originally had match key = email + last_name + property_address,
// but `contacts` has no property_address column — it lives on `loans`. We simplify
// to email + last_name; multi-match case is handled by the preview UI letting Scott
// pick which existing contact to attach to.
export async function findContactMatch(
  parsed: ParsedMismo,
  organizationId: string,
  sb: SupabaseClient,
): Promise<ContactMatch | null> {
  const email = parsed.borrower.email ? normalizeEmailForMatch(parsed.borrower.email) : null
  if (!email) return null

  const lastName = normalizeLastNameForMatch(parsed.borrower.last_name)

  const { data, error } = await sb
    .from('contacts')
    .select('id, first_name, last_name, email')
    .eq('organization_id', organizationId)
    .ilike('email', email)
    .ilike('last_name', lastName)
    .limit(5)

  if (error) throw new Error(`findContactMatch: ${error.message}`)
  if (!data?.length) return null

  const match = data[0]
  return {
    contact_id: match.id,
    first_name: match.first_name ?? null,
    last_name: match.last_name ?? '',
    email: match.email ?? null,
    property_address: null,
  }
}

export async function findLoanMatch(
  parsed: ParsedMismo,
  organizationId: string,
  sb: SupabaseClient,
): Promise<LoanMatch | null> {
  if (!parsed.loan.loan_number) return null

  const { data, error } = await sb
    .from('loans')
    .select('id, loan_number, borrower_name')
    .eq('organization_id', organizationId)
    .eq('loan_number', parsed.loan.loan_number)
    .maybeSingle()

  if (error) throw new Error(`findLoanMatch: ${error.message}`)
  if (!data) return null

  return {
    loan_id: data.id,
    loan_number: data.loan_number ?? '',
    borrower_name: data.borrower_name ?? null,
  }
}

const BUCKET = 'mismo-files'

export interface UploadResult {
  path: string
  fileName: string
}

/**
 * Upload a raw XML buffer to the pending folder of the user's org.
 * Returns the Storage path — stored transiently until commit, then moved.
 */
export async function uploadPending(
  sb: SupabaseClient,
  organizationId: string,
  xmlBuffer: Buffer | Uint8Array,
  uploadUuid: string,
): Promise<UploadResult> {
  const path = `${organizationId}/pending-${uploadUuid}.xml`
  const { error } = await sb.storage.from(BUCKET).upload(path, xmlBuffer, {
    contentType: 'application/xml',
    upsert: false,
  })
  if (error) throw new Error(`Storage upload failed: ${error.message}`)
  return { path, fileName: `pending-${uploadUuid}.xml` }
}

/**
 * Move a pending file to its canonical loan path on commit.
 * Returns the new path.
 */
export async function moveToCanonical(
  sb: SupabaseClient,
  organizationId: string,
  pendingPath: string,
  loanId: string,
): Promise<string> {
  const newPath = `${organizationId}/loan-${loanId}.xml`
  const { error } = await sb.storage.from(BUCKET).move(pendingPath, newPath)
  if (error) throw new Error(`Storage move failed: ${error.message}`)
  return newPath
}

/**
 * Delete a pending file (on parse failure or cancel).
 */
export async function deleteFile(sb: SupabaseClient, path: string): Promise<void> {
  const { error } = await sb.storage.from(BUCKET).remove([path])
  if (error) throw new Error(`Storage delete failed: ${error.message}`)
}

import { NextRequest, NextResponse } from 'next/server'
import { randomUUID } from 'node:crypto'
import { createServiceClient } from '@/lib/supabase/service'
import { getOrganization } from '@/lib/getOrganization'
import { parseMismo } from '@/lib/mismo/parser'
import { findContactMatch, findLoanMatch } from '@/lib/mismo/matcher'
import { uploadPending, deleteFile } from '@/lib/mismo/storage'

export const runtime = 'nodejs'
export const maxDuration = 30

const MAX_FILE_SIZE = 2 * 1024 * 1024 // 2 MB

export async function POST(req: NextRequest) {
  const { organizationId } = await getOrganization()
  const sb = createServiceClient()

  const formData = await req.formData()
  const file = formData.get('file')
  if (!(file instanceof File)) {
    return NextResponse.json({ error: { code: 'no_file', message: 'No file uploaded' } }, { status: 400 })
  }
  if (file.size > MAX_FILE_SIZE) {
    return NextResponse.json({ error: { code: 'too_large', message: 'File exceeds 2 MB limit' } }, { status: 400 })
  }

  const buffer = Buffer.from(await file.arrayBuffer())
  const xml = buffer.toString('utf-8')

  const uploadUuid = randomUUID()
  const { path: pendingPath } = await uploadPending(sb, organizationId, buffer, uploadUuid)

  const result = parseMismo(xml)
  if (!result.ok) {
    await deleteFile(sb, pendingPath).catch(() => { /* best-effort cleanup */ })
    return NextResponse.json({ error: result.error }, { status: 400 })
  }

  const [contactMatch, loanMatch] = await Promise.all([
    findContactMatch(result.data, organizationId, sb),
    findLoanMatch(result.data, organizationId, sb),
  ])

  return NextResponse.json({
    parsed: result.data,
    contactMatch,
    loanMatch,
    fileUrl: pendingPath,
  })
}

import { moveToCanonical } from '@/lib/mismo/storage'
import type { Database } from '@/lib/database.types'
import type { ParsedMismo } from '@/lib/mismo/types'

export const runtime = 'nodejs'
export const maxDuration = 30

type ContactInsert = Database['public']['Tables']['contacts']['Insert']
type LoanInsert = Database['public']['Tables']['loans']['Insert']

interface CommitBody {
  edited: ParsedMismo
  fileUrl: string // pending path
  contactMatchId: string | null // null = create new
  loanMatchId: string | null // null = create new
  updateContact: boolean // if true and contactMatchId set, update that row; if false, create new
}

export async function POST(req: NextRequest) {
  const { organizationId, userId } = await getOrganization()
  const sb = createServiceClient()

  const body = (await req.json()) as CommitBody
  const { edited, fileUrl, contactMatchId, loanMatchId, updateContact } = body

  // --- upsert contact ---
  // Contact-level columns only: the subject property address lives on `loans`, not `contacts`.
  let contactId: string
  const contactPayload: Partial<ContactInsert> = {
    organization_id: organizationId,
    first_name: edited.borrower.first_name ?? '',
    last_name: edited.borrower.last_name,
    email: edited.borrower.email,
    phone: edited.borrower.phone,
    birthdate: edited.borrower.dob,
    co_borrower_first: edited.co_borrower?.first_name ?? null,
    co_borrower_last: edited.co_borrower?.last_name ?? null,
    co_borrower_email: edited.co_borrower?.email ?? null,
    co_borrower_mobile: edited.co_borrower?.phone ?? null,
    co_borrower_birthdate: edited.co_borrower?.dob ?? null,
  }

  if (contactMatchId && updateContact) {
    const { error } = await sb.from('contacts').update(contactPayload).eq('id', contactMatchId).eq('organization_id', organizationId)
    if (error) return NextResponse.json({ error: { code: 'contact_update_failed', message: error.message } }, { status: 500 })
    contactId = contactMatchId
  } else {
    const { data, error } = await sb.from('contacts').insert(contactPayload as ContactInsert).select('id').single()
    if (error || !data) return NextResponse.json({ error: { code: 'contact_create_failed', message: error?.message ?? 'unknown' } }, { status: 500 })
    contactId = data.id
  }

  // --- upsert loan ---
  // Column names verified against src/lib/database.types.ts (2026-04-22).
  const loanPayload: Partial<LoanInsert> = {
    organization_id: organizationId,
    loan_number: edited.loan.loan_number,
    loan_amount: edited.loan.loan_amount,
    loan_purpose: edited.loan.loan_purpose,
    loan_type: edited.loan.loan_type,
    term_months: edited.loan.term_months,
    estimated_closing_date: edited.loan.estimated_close_date, // note column name difference
    property_address: edited.property.address,
    property_city: edited.property.city,
    property_state: edited.property.state,
    property_zip: edited.property.zip,
    property_value: edited.property.property_value,
    borrower_name: [edited.borrower.first_name, edited.borrower.last_name].filter(Boolean).join(' '),
    co_borrower_name: edited.co_borrower
      ? [edited.co_borrower.first_name, edited.co_borrower.last_name].filter(Boolean).join(' ')
      : null,
    co_borrower_email: edited.co_borrower?.email ?? null,
    co_borrower_phone: edited.co_borrower?.phone ?? null,
  }
  let loanId: string

  if (loanMatchId) {
    const { error } = await sb.from('loans').update(loanPayload).eq('id', loanMatchId).eq('organization_id', organizationId)
    if (error) return NextResponse.json({ error: { code: 'loan_update_failed', message: error.message } }, { status: 500 })
    loanId = loanMatchId
  } else {
    const { data, error } = await sb.from('loans').insert(loanPayload as LoanInsert).select('id').single()
    if (error || !data) return NextResponse.json({ error: { code: 'loan_create_failed', message: error?.message ?? 'unknown' } }, { status: 500 })
    loanId = data.id
  }

  // --- move Storage file to canonical location ---
  let canonicalPath: string
  try {
    canonicalPath = await moveToCanonical(sb, organizationId, fileUrl, loanId)
  } catch (e) {
    return NextResponse.json({ error: { code: 'storage_move_failed', message: (e as Error).message } }, { status: 500 })
  }

  // --- set loans.mismo_file_url ---
  const { error: urlErr } = await sb.from('loans').update({ mismo_file_url: canonicalPath }).eq('id', loanId)
  if (urlErr) return NextResponse.json({ error: { code: 'loan_url_update_failed', message: urlErr.message } }, { status: 500 })

  // --- activity_log ---
  // activity_log schema (verified 2026-04-22): `action` is required; `type`,
  // `event_type`, `summary`, `loan_id`, `contact_id`, `user_id`, `organization_id`
  // are optional. We use `action='imported'` + `type='mismo_import'` to keep the
  // timeline query distinguishable.
  type ActivityInsert = Database['public']['Tables']['activity_log']['Insert']
  const activityPayload: ActivityInsert = {
    action: loanMatchId ? 'updated' : 'imported',
    type: 'mismo_import',
    organization_id: organizationId,
    loan_id: loanId,
    contact_id: contactId,
    user_id: userId,
    summary: loanMatchId
      ? `MISMO file re-imported for loan #${edited.loan.loan_number ?? 'unknown'}`
      : `MISMO file imported for ${edited.borrower.first_name ?? ''} ${edited.borrower.last_name}`.trim(),
  }
  await sb.from('activity_log').insert(activityPayload)

  return NextResponse.json({ loanId })
}
```

'use client'

import { useState } from 'react'
import * as Dialog from '@radix-ui/react-dialog'
import { X } from 'lucide-react'
import { MismoPreviewForm } from './MismoPreviewForm'
import type { ParsedMismo, ContactMatch, LoanMatch, ParseError } from '@/lib/mismo/types'

interface Props {
  open: boolean
  onOpenChange: (open: boolean) => void
}

type Stage =
  | { kind: 'upload' }
  | { kind: 'parsing' }
  | { kind: 'error'; error: ParseError | { code: string; message: string } }
  | { kind: 'preview'; parsed: ParsedMismo; contactMatch: ContactMatch | null; loanMatch: LoanMatch | null; fileUrl: string }

const MAX_FILE_SIZE = 2 * 1024 * 1024

export function MismoImportModal({ open, onOpenChange }: Props) {
  const [stage, setStage] = useState<Stage>({ kind: 'upload' })

  async function handleFile(file: File) {
    if (!file.name.endsWith('.xml')) {
      setStage({ kind: 'error', error: { code: 'not_xml', message: "File doesn't look like XML. Export as MISMO 3.4 from Point." } })
      return
    }
    if (file.size > MAX_FILE_SIZE) {
      setStage({ kind: 'error', error: { code: 'too_large', message: 'File too large (max 2 MB).' } })
      return
    }

    setStage({ kind: 'parsing' })
    const form = new FormData()
    form.append('file', file)
    const res = await fetch('/api/mismo/parse', { method: 'POST', body: form })
    const body = await res.json()
    if (!res.ok) {
      setStage({ kind: 'error', error: body.error })
      return
    }
    setStage({ kind: 'preview', parsed: body.parsed, contactMatch: body.contactMatch, loanMatch: body.loanMatch, fileUrl: body.fileUrl })
  }

  return (
    <Dialog.Root open={open} onOpenChange={onOpenChange}>
      <Dialog.Portal>
        <Dialog.Overlay className="fixed inset-0 bg-black/40" />
        <Dialog.Content className="fixed left-1/2 top-1/2 w-[90vw] max-w-2xl -translate-x-1/2 -translate-y-1/2 rounded-lg bg-white p-6 shadow-xl">
          <div className="flex items-start justify-between">
            <Dialog.Title className="text-lg font-semibold">Import MISMO 3.4 file</Dialog.Title>
            <Dialog.Close asChild>
              <button className="text-gray-400 hover:text-gray-600"><X className="h-5 w-5" /></button>
            </Dialog.Close>
          </div>

          {stage.kind === 'upload' && <UploadDropzone onFile={handleFile} />}
          {stage.kind === 'parsing' && <p className="mt-4 text-sm text-gray-600">Parsing…</p>}
          {stage.kind === 'error' && (
            <div className="mt-4 rounded border border-red-200 bg-red-50 p-3 text-sm">
              <p className="font-medium text-red-800">{stage.error.message}</p>
              {'missingFields' in stage.error && stage.error.missingFields?.length ? (
                <ul className="mt-2 list-disc pl-5 text-red-700">
                  {stage.error.missingFields.map(f => <li key={f}>{f}</li>)}
                </ul>
              ) : null}
              <button className="mt-3 text-sm underline" onClick={() => setStage({ kind: 'upload' })}>Try another file</button>
            </div>
          )}
          {stage.kind === 'preview' && (
            <MismoPreviewForm
              parsed={stage.parsed}
              contactMatch={stage.contactMatch}
              loanMatch={stage.loanMatch}
              fileUrl={stage.fileUrl}
              onCancel={() => onOpenChange(false)}
            />
          )}
        </Dialog.Content>
      </Dialog.Portal>
    </Dialog.Root>
  )
}

function UploadDropzone({ onFile }: { onFile: (file: File) => void }) {
  return (
    <label className="mt-4 flex h-40 cursor-pointer items-center justify-center rounded-md border-2 border-dashed border-gray-300 text-sm text-gray-600 hover:border-gray-400">
      <span>Drop a .xml file here, or click to browse</span>
      <input
        type="file"
        accept=".xml,application/xml,text/xml"
        className="hidden"
        onChange={e => {
          const f = e.target.files?.[0]
          if (f) onFile(f)
        }}
      />
    </label>
  )
}

'use client'

import { useState } from 'react'
import { useRouter } from 'next/navigation'
import type { ParsedMismo, ContactMatch, LoanMatch } from '@/lib/mismo/types'

interface Props {
  parsed: ParsedMismo
  contactMatch: ContactMatch | null
  loanMatch: LoanMatch | null
  fileUrl: string
  onCancel: () => void
}

export function MismoPreviewForm({ parsed, contactMatch, loanMatch, fileUrl, onCancel }: Props) {
  const router = useRouter()
  const [edited, setEdited] = useState<ParsedMismo>(parsed)
  const [updateContact, setUpdateContact] = useState<boolean>(!!contactMatch)
  const [submitting, setSubmitting] = useState(false)
  const [error, setError] = useState<string | null>(null)

  function updateBorrower<K extends keyof ParsedMismo['borrower']>(key: K, value: ParsedMismo['borrower'][K]) {
    setEdited(e => ({ ...e, borrower: { ...e.borrower, [key]: value } }))
  }
  function updateProperty<K extends keyof ParsedMismo['property']>(key: K, value: ParsedMismo['property'][K]) {
    setEdited(e => ({ ...e, property: { ...e.property, [key]: value } }))
  }
  function updateLoan<K extends keyof ParsedMismo['loan']>(key: K, value: ParsedMismo['loan'][K]) {
    setEdited(e => ({ ...e, loan: { ...e.loan, [key]: value } }))
  }

  async function onSubmit(ev: React.FormEvent) {
    ev.preventDefault()
    setSubmitting(true)
    setError(null)
    const res = await fetch('/api/mismo/commit', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        edited,
        fileUrl,
        contactMatchId: contactMatch?.contact_id ?? null,
        loanMatchId: loanMatch?.loan_id ?? null,
        updateContact,
      }),
    })
    const body = await res.json()
    setSubmitting(false)
    if (!res.ok) {
      setError(body.error?.message ?? 'Unknown error')
      return
    }
    router.push(`/dashboard/pipeline/${body.loanId}`)
  }

  return (
    <form onSubmit={onSubmit} className="mt-4 space-y-4 text-sm">
      {contactMatch && (
        <label className="flex items-start gap-2 rounded border border-blue-200 bg-blue-50 p-3">
          <input type="checkbox" checked={updateContact} onChange={e => setUpdateContact(e.target.checked)} />
          <span>
            <strong>Match found:</strong> {contactMatch.first_name} {contactMatch.last_name} · {contactMatch.property_address}.
            Uncheck to create a new contact instead.
          </span>
        </label>
      )}
      {loanMatch && (
        <div className="rounded border border-amber-200 bg-amber-50 p-3">
          <strong>Existing loan #{loanMatch.loan_number}</strong> will be updated with any fields you edit below.
        </div>
      )}
      {!loanMatch && !edited.loan.loan_number && (
        <div className="rounded border border-amber-200 bg-amber-50 p-3">
          No loan number in file — this will create a new pipeline row. Set manually:
          <input
            type="text"
            className="ml-2 rounded border border-gray-300 px-2 py-1"
            onChange={e => updateLoan('loan_number', e.target.value || null)}
          />
        </div>
      )}

      <Section title="Borrower">
        <Field label="First name" value={edited.borrower.first_name ?? ''} onChange={v => updateBorrower('first_name', v || null)} />
        <Field label="Last name" value={edited.borrower.last_name} onChange={v => updateBorrower('last_name', v)} required />
        <Field label="Email" value={edited.borrower.email ?? ''} onChange={v => updateBorrower('email', v || null)} />
        <Field label="Phone" value={edited.borrower.phone ?? ''} onChange={v => updateBorrower('phone', v || null)} />
      </Section>

      <Section title="Property">
        <Field label="Street address" value={edited.property.address} onChange={v => updateProperty('address', v)} required />
        <Field label="City" value={edited.property.city} onChange={v => updateProperty('city', v)} required />
        <Field label="State" value={edited.property.state} onChange={v => updateProperty('state', v)} required />
        <Field label="ZIP" value={edited.property.zip} onChange={v => updateProperty('zip', v)} required />
        <Field label="Property value" value={String(edited.property.property_value ?? '')} onChange={v => updateProperty('property_value', v ? Number(v) : null)} />
      </Section>

      <Section title="Loan">
        <Field label="Loan amount" value={String(edited.loan.loan_amount)} onChange={v => updateLoan('loan_amount', Number(v))} required />
        <Field label="Purpose" value={edited.loan.loan_purpose} onChange={v => updateLoan('loan_purpose', v as ParsedMismo['loan']['loan_purpose'])} required />
        <Field label="Term (months)" value={String(edited.loan.term_months ?? '')} onChange={v => updateLoan('term_months', v ? Number(v) : null)} />
        <Field label="Estimated close date" value={edited.loan.estimated_close_date ?? ''} onChange={v => updateLoan('estimated_close_date', v || null)} />
      </Section>

      {error && <div className="rounded border border-red-200 bg-red-50 p-3 text-red-800">{error}</div>}

      <div className="flex justify-end gap-2">
        <button type="button" onClick={onCancel} className="rounded border border-gray-300 px-3 py-1.5">Cancel</button>
        <button type="submit" disabled={submitting} className="rounded bg-blue-600 px-3 py-1.5 text-white disabled:opacity-50">
          {submitting ? 'Saving…' : 'Confirm import'}
        </button>
      </div>
    </form>
  )
}

function Section({ title, children }: { title: string; children: React.ReactNode }) {
  return (
    <fieldset className="rounded border border-gray-200 p-3">
      <legend className="px-1 text-xs font-semibold uppercase text-gray-500">{title}</legend>
      <div className="mt-2 grid grid-cols-2 gap-3">{children}</div>
    </fieldset>
  )
}

function Field({ label, value, onChange, required }: { label: string; value: string; onChange: (v: string) => void; required?: boolean }) {
  return (
    <label className="block">
      <span className="text-xs text-gray-600">{label}{required && <span className="text-red-500">*</span>}</span>
      <input
        type="text"
        value={value}
        onChange={e => onChange(e.target.value)}
        required={required}
        className="mt-1 block w-full rounded border border-gray-300 px-2 py-1"
      />
    </label>
  )
}
